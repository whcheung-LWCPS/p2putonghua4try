        function playSound(type) {

            if (isMuted) return;

            try {

                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

                if (audioCtx.state === 'suspended') audioCtx.resume();


                const osc = audioCtx.createOscillator();

                const gain = audioCtx.createGain();

                osc.connect(gain);

                gain.connect(audioCtx.destination);


                const now = audioCtx.currentTime;


                if (type === 'correct') {

                    // 答對雙提示音

                    osc.type = 'sine';

                    osc.frequency.setValueAtTime(523.25, now);

                    osc.frequency.setValueAtTime(659.25, now + 0.12);

                    gain.gain.setValueAtTime(0.3, now);

                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);

                    osc.start(now);

                    osc.stop(now + 0.4);

                } else if (type === 'wrong') {

                    // 答錯低音

                    osc.type = 'triangle';

                    osc.frequency.setValueAtTime(220, now);

                    osc.frequency.setValueAtTime(174.61, now + 0.15);

                    gain.gain.setValueAtTime(0.3, now);

                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);

                    osc.start(now);

                    osc.stop(now + 0.35);

                }

            } catch (e) {

                console.log("Audio play error:", e);

            }

        }

        function pcmToWav(pcm16Array, sampleRate) {

            const numChannels = 1;

            const bytesPerSample = 2;

            const blockAlign = numChannels * bytesPerSample;

            const byteRate = sampleRate * blockAlign;

            const dataSize = pcm16Array.length * bytesPerSample;

            const buffer = new ArrayBuffer(44 + dataSize);

            const view = new DataView(buffer);


            const writeString = (v, offset, str) => {

                for (let i = 0; i < str.length; i++) v.setUint8(offset + i, str.charCodeAt(i));

            };


            writeString(view, 0, 'RIFF');

            view.setUint32(4, 36 + dataSize, true);

            writeString(view, 8, 'WAVE');

            writeString(view, 12, 'fmt ');

            view.setUint32(16, 16, true);

            view.setUint16(20, 1, true);

            view.setUint16(22, numChannels, true);

            view.setUint32(24, sampleRate, true);

            view.setUint32(28, byteRate, true);

            view.setUint16(32, blockAlign, true);

            view.setUint16(34, 16, true);

            writeString(view, 36, 'data');

            view.setUint32(40, dataSize, true);


            let offset = 44;

            for (let i = 0; i < pcm16Array.length; i++, offset += 2) {

                view.setInt16(offset, pcm16Array[i], true);

            }


            return new Blob([buffer], { type: 'audio/wav' });

        }


        function base64ToArrayBuffer(base64) {

            const binaryString = window.atob(base64);

            const len = binaryString.length;

            const bytes = new Uint8Array(len);

            for (let i = 0; i < len; i++) {

                bytes[i] = binaryString.charCodeAt(i);

            }

            return bytes.buffer;

        }


        async function speakWithGeminiTTS(text) {

            try {

                const apiKey = "";

                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;

                

                // 直接傳送純中文，確保 AI 自動識別並以標準普通話讀出

                const payload = {

                    contents: [{

                        parts: [{ text: text }]

                    }],

                    generationConfig: {

                        responseModalities: ["AUDIO"],

                        speechConfig: {

                            voiceConfig: {

                                prebuiltVoiceConfig: { voiceName: "Sulafat" } // 暖心標準國語女聲

                            }

                        }

                    }

                };


                const res = await fetch(apiUrl, {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify(payload)

                });


                if (!res.ok) return false;


                const data = await res.json();

                const part = data.candidates?.[0]?.content?.parts?.[0];

                const audioData = part?.inlineData?.data;

                const mimeType = part?.inlineData?.mimeType || "";


                if (audioData) {

                    let sampleRate = 24000;

                    const match = mimeType.match(/rate=(\d+)/);

                    if (match && match[1]) sampleRate = parseInt(match[1], 10);


                    const pcmBuffer = base64ToArrayBuffer(audioData);

                    const pcm16 = new Int16Array(pcmBuffer);

                    const wavBlob = pcmToWav(pcm16, sampleRate);

                    const audioUrl = URL.createObjectURL(wavBlob);

                    

                    const audio = new Audio(audioUrl);

                    await new Promise((resolve) => {

                        audio.onended = () => resolve(true);

                        audio.onerror = () => resolve(false);

                        audio.play().catch(() => resolve(false));

                    });

                    return true;

                }

            } catch (e) {

                console.warn("Gemini TTS API error:", e);

            }

            return false;

        }


        function speakWithGoogleTTS(text) {

            return new Promise((resolve) => {

                try {

                    const url = `https://translate.google.com/translate_tts?ie=UTF-8&q=${encodeURIComponent(text)}&tl=zh-CN&client=tw-ob`;

                    const audio = new Audio(url);

                    audio.onended = () => resolve(true);

                    audio.onerror = () => resolve(false);

                    audio.play().then(() => {}).catch(() => resolve(false));

                } catch(e) {

                    resolve(false);

                }

            });

        }


        function getMandarinVoice() {

            if (!('speechSynthesis' in window)) return null;

            const voices = window.speechSynthesis.getVoices();

            if (!voices || voices.length === 0) return null;


            // 嚴格檢測並排除廣東話 / 粵語語音包

            const isCantonese = (v) => {

                const lang = (v.lang || '').toLowerCase();

                const name = (v.name || '').toLowerCase();

                return lang.includes('hk') || lang.includes('yue') || name.includes('cantonese') || name.includes('hong kong') || name.includes('sin-ji');

            };


            // 第一優先：標準 zh-CN / zh-Hans

            let v = voices.find(v => !isCantonese(v) && (v.lang.toLowerCase() === 'zh-cn' || v.lang.toLowerCase() === 'zh_cn' || v.lang.toLowerCase() === 'zh-hans-cn'));

            

            // 第二優先：知名普通話語音庫名稱 (TingTing, Yaoyao, HuiHui, Putonghua)

            if (!v) {

                const kw = ['mandarin', 'putonghua', 'tingting', 'yaoyao', 'huihui', 'kangkang', 'cmn', 'zh-hans'];

                v = voices.find(v => !isCantonese(v) && kw.some(k => v.lang.toLowerCase().includes(k) || v.name.toLowerCase().includes(k)));

            }


            // 第三優先：國語 zh-TW 台灣國語（比粵語更符合普通話教學需要）

            if (!v) {

                v = voices.find(v => !isCantonese(v) && (v.lang.toLowerCase().includes('tw') || v.name.toLowerCase().includes('taiwan')));

            }


            return v;

        }


        function speakWithWebSpeechMandarin(text) {

            if (!('speechSynthesis' in window)) {

                hideAudioIndicator();

                return;

            }


            window.speechSynthesis.cancel();

            const utterance = new SpeechSynthesisUtterance(text);

            utterance.lang = 'zh-CN';

            utterance.rate = 0.85;


            const mandarinVoice = getMandarinVoice();

            if (mandarinVoice) {

                utterance.voice = mandarinVoice;

                utterance.onend = hideAudioIndicator;

                utterance.onerror = hideAudioIndicator;

                window.speechSynthesis.speak(utterance);

            } else {

                // 如果裝置完全沒有普通話語音庫，寧可不發聲也不播放廣東話，並於1.2秒後隱藏指示標籤

                console.warn("裝置暫無普通話語音庫，已自動封鎖廣東話播放。");

                setTimeout(hideAudioIndicator, 1200);

            }

        }


        async function speakMandarin(text, event) {

            if (event) event.stopPropagation();


            showAudioIndicator(text);


            // 第一層：Gemini AI TTS 標準國語

            const geminiSuccess = await speakWithGeminiTTS(text);

            if (geminiSuccess) {

                hideAudioIndicator();

                return;

            }


            // 第二層：線上標準普通話音訊

            const onlineSuccess = await speakWithGoogleTTS(text);

            if (onlineSuccess) {

                hideAudioIndicator();

                return;

            }


            // 第三層：Web Speech API（嚴格過濾並排除廣東話）

            speakWithWebSpeechMandarin(text);

        }
