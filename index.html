function speakText(text) { if (!isAudioEnabled) return; unlockAudioOnIOS(); speakTextWebSpeechFallback(text); } // 尋找普通話語音，排除廣東話 function findMandarinVoice() { const synth = window.speechSynthesis; const voices = synth.getVoices() || []; const mandarinVoices = voices.filter(voice => { const lang = (voice.lang || '') .toLowerCase() .replace(/_/g, '-'); const name = (voice.name || '').toLowerCase(); // 排除廣東話 const isCantonese = lang.includes('zh-hk') || lang.includes('yue') || name.includes('cantonese') || name.includes('sin-ji') || name.includes('sinji') || name.includes('廣東話') || name.includes('粵語'); if (isCantonese) return false; // 識別普通話 return ( /^zh-(cn|sg|tw|cmn)(-|$)/.test(lang) || name.includes('mandarin') || name.includes('putonghua') || name.includes('普通話') || name.includes('國語') || name.includes('ting-ting') || name.includes('tingting') || name.includes('mei-jia') ); }); // 優先選擇中國大陸普通話 return ( mandarinVoices.find(v => v.lang.toLowerCase().replace(/_/g, '-') === 'zh-cn' ) || mandarinVoices.find(v => v.lang.toLowerCase().startsWith('zh-sg') ) || mandarinVoices[0] || null ); } // 普通話朗讀 function speakTextWebSpeechFallback(text) { const synth = window.speechSynthesis; if (!synth) { alert('此 iPad 不支援語音朗讀。'); return; } synth.cancel(); const voice = findMandarinVoice(); // 沒有普通話聲線就不朗讀，防止變成廣東話 if (!voice) { alert( '這部 iPad 找不到普通話語音。' + '請在 iPad 設定中下載普通話聲線，然後重新打開網頁。' ); console.warn( 'Available voices:', synth.getVoices().map(v => ({ name: v.name, lang: v.lang })) ); return; } const utterance = new SpeechSynthesisUtterance(text); // 明確指定普通話聲線 utterance.voice = voice; utterance.lang = voice.lang; utterance.rate = 0.85; utterance.pitch = 1; utterance.volume = 1; synth.speak(utterance); }
